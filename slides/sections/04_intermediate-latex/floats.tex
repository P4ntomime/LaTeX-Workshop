\begin{frame}
    \frametitle{Positioning Figures and Tables}
    \begin{description}
        \item<+->[Problem] \lstinline|\\includegraphics[]\{\}| and \lstinline|\\begin\{tabular\}...\\end\{tabular\}| typeset images/tables inline
        \begin{itemize}
            \item<+-> Often leads to ugly documents
            \item<+-> Captions can be tedious
        \end{itemize}
        \item<+->[Idea] Only give \LaTeX{} a placement \emph{suggestion} and let it decide where to optimally place the image.
        \item<+->[Solution] Floating environments \texttt{figure} and \texttt{table}.
    \end{description}
\end{frame}
\begin{frame}
    \frametitle{Floating Environments}
    \framesubtitle<-7>{Structure}
    \only<-7>{%
    \Large
    \texttt{\bslash\lstkw{begin}\bropen figure\brclose\uncover<2->{\tn{placement}{[t]}}}\\
    \uncover<3->{\texttt{\phantom{xx}\tn{alignment}{\bslash\lstkw{centering}}}\\}
    \uncover<4->{\texttt{\phantom{xx}\bslash\lstkw{includegraphics}\bropen example-image-a\brclose}\\}
    \uncover<5->{\texttt{\phantom{xx}\tn{caption}{\bslash\lstkw{caption}\bropen A caption\brclose}}\\}
    \uncover<6->{\texttt{\phantom{xx}\tn{label}{\bslash\lstkw{label}\bropen fig:example-a\brclose}}\\}
    \texttt{\bslash\lstkw{end}\bropen figure\brclose}

    \begin{tikzpicture}[thick,remember picture, overlay, every node/.style={inner xsep=0.5pt, inner ysep=1pt}]
        \coordinate[right=5cm of placement] (txtleft);
        \only<2,7>{%

        \only<3,7>{%
        \node[draw, dashed, inner xsep=3pt,fit={(alignment)}] (alignmentoutline){};
