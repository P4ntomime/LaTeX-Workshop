\begin{frame}
    \frametitle{Lists}
    % We have 3 different kinds of lists: itemize, enumerate, description
    \centering
    \begin{tabular}{c l}
        \textbf{Environment}
        & \textbf{Description}\\
    \toprule[1.2pt]\pause
        \texttt{itemize}
        & For simple lists with bullet points.\\
    \midrule\pause
        \texttt{enumerate}
        & For numbered lists.\\
    \midrule\pause
        \texttt{description}
        & For lists describing terms.\\
    \end{tabular}

    \only<5->{%
    \begin{center}
        \begin{myblock}[width=16em,middle=0mm,valign lower=top,valign=center]{\blockstrut Special Command}
            \lstinline|\\item[]|
        \tcblower
        \begin{itemize}
            \item Delimits a new item.
            \item \lstinline|[]| is optional for \texttt{itemize} and \texttt{enumerate}.
        \end{itemize}
        \end{myblock}
    \end{center}}
\end{frame}
